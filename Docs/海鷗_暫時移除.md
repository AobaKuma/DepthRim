# 海鷗（裝飾性生態演出）— 暫時移除，日後再加回

海面兩層的裝飾性海鷗：飛來 → 降落停留 → 被嚇飛或時間到飛走。純畫面，不是 Pawn、不存檔。
已在 Release 建置、單元測試與驗收測試 K2（31 項）全數通過後才移除；程式碼原樣保存在本文末尾，照「加回步驟」貼回即可。

## 設計摘要

- **生成**：每層每 250 tick 判斷一次，機率 1 − e^(−250/5000) ≈ 4.9%（平均 2 遊戲小時一次，一倍速約 83 秒）。25% 機率是 2～3 隻一小群（半徑 3.9 格內）。
- **上限**：每層同時最多 5 隻（`SeagullMath.MaxPerMap`）。只在「正在看的那一層」生成，且模組設定 `seagulls` 開啟、`Applies` 成立。
- **條件**：天光 ≥ 0.35（夜裡不來）、`SeaSway` 天氣係數 ≤ 2（暴風雨不來）。
- **落腳**（`SeagullMath.PerchFor`）：
  - 平台層：只站在 `DepthRim_OceanPlatform` 甲板上。
  - 水面層：浮在 `DepthRim_OpenSea`，或站在 `HullPlatformPropsExtension.FloatsOnSea` 的船體平台上。
  - 不落在高空（`DepthRim_OpenAir`）、支柱/建築、屋頂下、有人的格子、平台甲板正下方的陰影。
- **停留與嚇飛**：停 1800～6000 tick；每 30 tick 檢查，3 格內有人或落腳處消失就起飛。
- **飛行**：巡航高度 3.5、速度 0.12 格/tick；從畫面外飛來（最後 40% 滑翔下降），起飛後 35% 時間爬升，飛到畫面外（18～70 格）。8 格拍翅影格、每格 2 tick，滑翔時減半。
- **繪製**：`MapDrawLayer_Seagulls`（非 section 的 MapDrawLayer，會被反射建立在每張地圖上，同 `MapDrawLayer_Midwater`），每幀 `Graphics.DrawMesh`。飛行中身體在 Skyfaller 高度、往北偏移高度量，地面畫 SkyfallerShadowCircle 影子（隨高度縮小）；停著用 Pawn 高度；浮水的跟著 `SeaSway.Offset` 漂。西向用翻面 mesh。
- **除錯動作**（DepthRim 分類）：「Seagull lands here」、「Seagull flock arrives」。
- **已修過的 bug**：成群生成時第一隻沒檢查上限 → `TrySpawnFlock` 開頭 `Room == 0` 就 return。

````csharp
using UnityEngine;

namespace DepthRim
{
    /// <summary>海鷗能落在哪種格子上。</summary>
    public enum SeagullPerch
    {
        /// <summary>不能落。</summary>
        None,

        /// <summary>站著：平台層的海上平台甲板、水面層的船體平台。</summary>
        Stand,

        /// <summary>浮在水面：水面層的開放海面。</summary>
        Float,
    }

    /// <summary>
    /// 海鷗（海面兩層的裝飾性生態演出，<see cref="MapComponent_Seagulls"/>）的純函式與參數：
    /// 生成機率、數量上限、哪些格子能落、飛行路線、高度與動畫影格。不碰遊戲物件，可單元測試。
    /// </summary>
    public static class SeagullMath
    {
        /// <summary>多少 tick 判斷一次要不要來一隻（或一小群）。</summary>
        public const int CheckInterval = 250;

        /// <summary>
        /// 平均多少遊戲小時來一次（一次一隻或一小群）。每次判斷的機率是 <see cref="ChancePerCheck"/>：
        /// 2 小時 = 5000 tick，每 250 tick 約 4.9%；一倍速下平均每 83 秒一次。
        /// </summary>
        public const float ArrivalMtbHours = 2f;

        /// <summary>每張地圖（每一層各自算）同時最多幾隻。</summary>
        public const int MaxPerMap = 5;

        /// <summary>一次來兩三隻的機率。</summary>
        public const float FlockChance = 0.25f;

        /// <summary>一小群裡的同伴落在離第一隻幾格以內。</summary>
        public const float FlockRadius = 3.9f;

        /// <summary>巡航高度（格，畫面上往北偏、影子留在地上）。</summary>
        public const float CruiseHeight = 3.5f;

        /// <summary>飛行速度（格／tick）：0.12 = 每秒 7.2 格。</summary>
        public const float FlightSpeed = 0.12f;

        /// <summary>路程最後多少比例是下降滑翔（之前維持巡航高度）。</summary>
        public const float GlideShare = 0.4f;

        /// <summary>起飛後多少比例的路程爬升到巡航高度。</summary>
        public const float ClimbShare = 0.35f;

        /// <summary>停留多久（tick）：30～100 秒（一倍速）。</summary>
        public const int MinRestTicks = 1800;

        public const int MaxRestTicks = 6000;

        /// <summary>有人走到幾格內就嚇飛。</summary>
        public const float DisturbRadius = 3f;

        /// <summary>停著的時候多久檢查一次有沒有人靠近、落腳處還在不在（tick）。</summary>
        public const int RestCheckInterval = 30;

        /// <summary>飛來／飛走的路程至少幾格；起點與終點盡量落在鏡頭畫面外。</summary>
        public const float MinFlightDistance = 18f;

        public const float MaxFlightDistance = 70f;

        /// <summary>起點／終點要離畫面邊緣多遠才算在畫面外（格）。</summary>
        public const float ViewMargin = 3f;

        /// <summary>天色太暗（夜裡）不來：天光低於這個值。</summary>
        public const float MinSkyGlow = 0.35f;

        /// <summary>風浪太大（暴風雨）不來：海面擺動的天氣倍率高於這個值（<see cref="SeaSway.WeatherFactor"/>）。</summary>
        public const float MaxSwayFactor = 2f;

        /// <summary>挑落腳格最多試幾次。</summary>
        public const int CellTries = 24;

        /// <summary>每次判斷來一隻的機率：1 - e^(-間隔 / 平均間隔)。</summary>
        public static float ChancePerCheck(float mtbHours = ArrivalMtbHours, int interval = CheckInterval)
        {
            if (mtbHours <= 0f)
            {
                return 1f;
            }
            return 1f - Mathf.Exp(-interval / (mtbHours * 2500f));
        }

        /// <summary>天色與天氣允許海鷗來嗎（夜裡、暴風雨不來）。</summary>
        public static bool WeatherAllows(float skyGlow, float swayFactor)
        {
            return skyGlow >= MinSkyGlow && swayFactor <= MaxSwayFactor;
        }

        /// <summary>還能再來幾隻（上限 <see cref="MaxPerMap"/>）。</summary>
        public static int Room(int current, int max = MaxPerMap)
        {
            return Mathf.Max(0, max - current);
        }

        /// <summary>
        /// 一格能不能落、怎麼落。平台層：只有海上平台甲板（站）。水面層：開放海面（浮）或浮在水上的船體平台（站），
        /// 但正上方有平台層甲板的地方不落（在高聳平台的陰影底下）。有建築、有屋頂、有人的格子都不落。
        /// </summary>
        public static SeagullPerch PerchFor(bool platformLevel, bool deck, bool openSea, bool floatingHull, bool platformAbove, bool edifice, bool roofed, bool occupied)
        {
            if (edifice || roofed || occupied)
            {
                return SeagullPerch.None;
            }
            if (platformLevel)
            {
                return deck ? SeagullPerch.Stand : SeagullPerch.None;
            }
            if (platformAbove)
            {
                return SeagullPerch.None;
            }
            if (openSea)
            {
                return SeagullPerch.Float;
            }
            return floatingHull ? SeagullPerch.Stand : SeagullPerch.None;
        }

        /// <summary>路程需要幾 tick（至少 1）。</summary>
        public static int TravelTicks(float distance, float speedPerTick = FlightSpeed)
        {
            return Mathf.Max(1, Mathf.CeilToInt(distance / Mathf.Max(speedPerTick, 1e-4f)));
        }

        /// <summary>
        /// 從 <paramref name="perch"/> 沿 <paramref name="dir"/> 走多遠才會離開畫面（<paramref name="viewMin"/>..<paramref name="viewMax"/> 外擴 <see cref="ViewMargin"/>），
        /// 夾在 <see cref="MinFlightDistance"/>..<see cref="MaxFlightDistance"/>。飛來時起點是 perch - dir × 這個距離，飛走時終點是 perch + dir × 這個距離。
        /// </summary>
        public static float DistanceToLeaveView(Vector2 perch, Vector2 dir, Vector2 viewMin, Vector2 viewMax)
        {
            if (dir.sqrMagnitude < 1e-8f)
            {
                return MinFlightDistance;
            }
            dir.Normalize();
            float minX = viewMin.x - ViewMargin, minZ = viewMin.y - ViewMargin, maxX = viewMax.x + ViewMargin, maxZ = viewMax.y + ViewMargin;
            bool inside = perch.x >= minX && perch.x <= maxX && perch.y >= minZ && perch.y <= maxZ;
            float d = inside ? MidwaterMath.DistanceToRectEdge(perch, dir, minX, minZ, maxX, maxZ) : 0f;
            // 飛行中畫在地面位置北方巡航高度處：往南飛出畫面時身體還在畫面裡，多算這一段
            if (dir.y < 0f)
            {
                d += CruiseHeight / Mathf.Max(-dir.y, 0.2f);
            }
            return Mathf.Clamp(d, MinFlightDistance, MaxFlightDistance);
        }

        /// <summary>飛來時的高度（格）：前段維持巡航高度，最後 <see cref="GlideShare"/> 滑翔降落（越接近越緩）。</summary>
        public static float ArrivalHeight(float progress)
        {
            progress = Mathf.Clamp01(progress);
            float glideStart = 1f - GlideShare;
            if (progress <= glideStart)
            {
                return CruiseHeight;
            }
            float g = (progress - glideStart) / GlideShare; // 0..1
            float remain = 1f - g;
            return CruiseHeight * remain * remain * (3f - 2f * remain); // smoothstep：落地前慢下來
        }

        /// <summary>飛走時的高度（格）：一開始先爬升，<see cref="ClimbShare"/> 之後維持巡航高度。</summary>
        public static float DepartureHeight(float progress)
        {
            progress = Mathf.Clamp01(progress);
            if (progress >= ClimbShare)
            {
                return CruiseHeight;
            }
            float g = progress / ClimbShare;
            return CruiseHeight * Mathf.Sqrt(g); // 起飛時爬得快
        }

        /// <summary>
        /// 拍翅的影格（0 起算）：飛行時每 <paramref name="ticksPerFrame"/> tick 換一格；滑翔時慢一倍（翅膀張開滑行）。
        /// </summary>
        public static int FlightFrame(int ageTicks, int frameCount, int ticksPerFrame, bool gliding)
        {
            if (frameCount <= 0)
            {
                return 0;
            }
            int perFrame = Mathf.Max(1, ticksPerFrame * (gliding ? 2 : 1));
            int f = (ageTicks / perFrame) % frameCount;
            return f < 0 ? f + frameCount : f;
        }

        /// <summary>
        /// 依前進方向挑貼圖方向：0 = 東、1 = 西（東翻面）、2 = 北、3 = 南。左右明顯時用側面（鳥飛行大多看得到側面）。
        /// </summary>
        public static int FacingIndex(Vector2 dir)
        {
            if (Mathf.Abs(dir.x) >= Mathf.Abs(dir.y) * 0.6f)
            {
                return dir.x >= 0f ? 0 : 1;
            }
            return dir.y >= 0f ? 2 : 3;
        }

        /// <summary>影子的大小倍率：越高越小（地面 1，巡航高度約 0.55）。</summary>
        public static float ShadowScale(float height)
        {
            return Mathf.Lerp(1f, 0.55f, Mathf.Clamp01(height / CruiseHeight));
        }
    }
}
````

### `.source/DepthRim/SeaSurface/Seagulls.cs`

````csharp
using System.Collections.Generic;
using LudeonTK;
using RimWorld;
using UnityEngine;
using Verse;

namespace DepthRim
{
    /// <summary>
    /// 一種海鷗的外觀。原版與 DLC 都沒有海鷗，預設借 Odyssey 天鵝（白色、有飛行影格與浮水姿勢）縮小來畫；
    /// 換成專屬貼圖時只要改這裡的路徑（見 1.7/Defs/SeaSurface/Seagulls.xml 的說明）。
    /// </summary>
    public class SeagullDef : Def
    {
        /// <summary>站著的樣子（Graphic_Multi：_east、_north、_south，西邊用東邊翻面）。</summary>
        public string standTexPath;

        /// <summary>浮在水面的樣子（Graphic_Multi）。空白就用站著的樣子。</summary>
        public string floatTexPath;

        /// <summary>飛行影格的路徑前綴：<c>前綴 + 1..N + _east/_north/_south</c>（和原版 PawnKindDef.flyingAnimationFramePathPrefix 同一種命名）。</summary>
        public string flyingFramePathPrefix;

        public int flyingFrameCount = 8;

        public int flyingTicksPerFrame = 2;

        public float standDrawSize = 0.9f;

        public float flyingDrawSize = 1.3f;

        /// <summary>出現的權重（有多種外觀時）。</summary>
        public float weight = 1f;

        [Unsaved(false)]
        private Graphic standGraphic;

        [Unsaved(false)]
        private Graphic floatGraphic;

        [Unsaved(false)]
        private Graphic[,] frames;

        [Unsaved(false)]
        private bool resolved;

        /// <summary>貼圖都找得到（主執行緒、第一次畫的時候才載入）。</summary>
        public bool Usable
        {
            get
            {
                Resolve();
                return standGraphic != null && frames != null;
            }
        }

        public Graphic StandGraphic
        {
            get
            {
                Resolve();
                return standGraphic;
            }
        }

        public Graphic FloatGraphic
        {
            get
            {
                Resolve();
                return floatGraphic ?? standGraphic;
            }
        }

        /// <summary>第 <paramref name="frame"/> 格飛行影格；<paramref name="dir"/>：0 = 東、1 = 北、2 = 南。</summary>
        public Graphic FlyingFrame(int frame, int dir)
        {
            Resolve();
            if (frames == null)
            {
                return null;
            }
            return frames[Mathf.Clamp(frame, 0, frames.GetLength(0) - 1), Mathf.Clamp(dir, 0, 2)];
        }

        private void Resolve()
        {
            if (resolved)
            {
                return;
            }
            resolved = true;
            Vector2 stand = new Vector2(standDrawSize, standDrawSize);
            if (Exists(standTexPath + "_east"))
            {
                standGraphic = GraphicDatabase.Get<Graphic_Multi>(standTexPath, ShaderDatabase.Cutout, stand, Color.white);
            }
            if (!floatTexPath.NullOrEmpty() && Exists(floatTexPath + "_east"))
            {
                floatGraphic = GraphicDatabase.Get<Graphic_Multi>(floatTexPath, ShaderDatabase.Cutout, stand, Color.white);
            }
            if (flyingFrameCount > 0 && Exists(flyingFramePathPrefix + "1_east"))
            {
                string[] suffix = { "_east", "_north", "_south" };
                Vector2 fly = new Vector2(flyingDrawSize, flyingDrawSize);
                Graphic[,] f = new Graphic[flyingFrameCount, 3];
                for (int i = 0; i < flyingFrameCount; i++)
                {
                    for (int d = 0; d < 3; d++)
                    {
                        string path = flyingFramePathPrefix + (i + 1) + suffix[d];
                        // 缺北／南影格時用東邊的
                        if (!Exists(path))
                        {
                            path = flyingFramePathPrefix + (i + 1) + "_east";
                        }
                        f[i, d] = GraphicDatabase.Get<Graphic_Single>(path, ShaderDatabase.Cutout, fly, Color.white);
                    }
                }
                frames = f;
            }
            if (standGraphic == null || frames == null)
            {
                Log.WarningOnce($"[DepthRim] seagull {defName}: textures not found ({standTexPath}, {flyingFramePathPrefix}); seagulls of this kind will not appear.", defName.GetHashCode() ^ 0x6011);
            }
        }

        private static bool Exists(string path)
        {
            return !path.NullOrEmpty() && ContentFinder<Texture2D>.Get(path, reportFailure: false) != null;
        }

        public override IEnumerable<string> ConfigErrors()
        {
            foreach (string e in base.ConfigErrors())
            {
                yield return e;
            }
            if (standTexPath.NullOrEmpty())
            {
                yield return "standTexPath is empty";
            }
            if (flyingFramePathPrefix.NullOrEmpty() || flyingFrameCount <= 0)
            {
                yield return "no flying frames";
            }
            if (standDrawSize <= 0f || flyingDrawSize <= 0f)
            {
                yield return "draw sizes must be positive";
            }
        }
    }

    public enum SeagullState
    {
        Arriving,
        Resting,
        Departing,
        Gone,
    }

    /// <summary>一隻海鷗（不是 Pawn、不存檔，純畫面）。</summary>
    public class Seagull
    {
        public SeagullDef def;

        public SeagullState state;

        /// <summary>落腳格。</summary>
        public IntVec3 cell;

        public SeagullPerch perch;

        /// <summary>落腳點（格內稍微偏一點，不會每隻都站正中央）。</summary>
        public Vector2 perchPos;

        /// <summary>這一段飛行的起點與終點（地面位置）。</summary>
        public Vector2 from;

        public Vector2 to;

        public int legTicks = 1;

        /// <summary>這一段（飛來、停留、飛走）已經過了幾 tick。</summary>
        public int stateAge;

        public int restTicks;

        /// <summary>停著時面向哪邊。</summary>
        public Rot4 facing = Rot4.East;

        /// <summary>動畫相位（拍翅、浮動不要每隻同步）。</summary>
        public int phase;

        /// <summary>給驗收測試：被人嚇飛的。</summary>
        public bool startled;

        public float Progress => Mathf.Clamp01(stateAge / (float)Mathf.Max(1, legTicks));

        public bool Flying => state == SeagullState.Arriving || state == SeagullState.Departing;

        /// <summary>地面位置（影子所在）。</summary>
        public Vector2 GroundPos
        {
            get
            {
                if (state == SeagullState.Resting)
                {
                    return perchPos;
                }
                return Vector2.Lerp(from, to, Progress);
            }
        }

        /// <summary>離地高度（格）。</summary>
        public float Height
        {
            get
            {
                switch (state)
                {
                    case SeagullState.Arriving:
                        return SeagullMath.ArrivalHeight(Progress);
                    case SeagullState.Departing:
                        return SeagullMath.DepartureHeight(Progress);
                    default:
                        return 0f;
                }
            }
        }

        public bool Gliding => state == SeagullState.Arriving && Progress > 1f - SeagullMath.GlideShare;
    }

    /// <summary>
    /// 海面兩層（平台層的平台頂、水面層）的海鷗：偶爾飛來一隻或一小群，降落在平台甲板或浮在海面上，停一陣子再飛走；有人靠近就嚇飛。
    ///
    /// <para><b>純畫面</b>：海鷗不是 Pawn，不存檔、不走尋路、不佔格子，畫面交給 <see cref="MapDrawLayer_Seagulls"/>。
    /// 只有目前正在看的那一層會來新的海鷗；不是海面地圖時每 tick 只判斷一次就離開。</para>
    ///
    /// <para><b>頻率與上限</b>：見 <see cref="SeagullMath"/>（每 250 tick 判斷一次，平均每 2 遊戲小時來一次，每層最多 5 隻，夜裡與暴風雨不來）。
    /// 模組設定可關（<see cref="DepthRimModSettings.seagulls"/>）。</para>
    /// </summary>
    public class MapComponent_Seagulls : MapComponent
    {
        private readonly List<Seagull> gulls = new List<Seagull>();

        private int checkOffset = -1;

        /// <summary>給驗收測試：來過幾隻、被嚇飛幾隻。</summary>
        public int totalArrivals;

        public int totalStartled;

        public MapComponent_Seagulls(Map map) : base(map)
        {
        }

        public IReadOnlyList<Seagull> Gulls => gulls;

        public int Count => gulls.Count;

        /// <summary>這張地圖是海面的一層（水面層或平台層）。</summary>
        public bool Applies => SeaLevels.IsAnyLevel(map);

        public override void MapComponentTick()
        {
            if (gulls.Count > 0)
            {
                Advance();
            }
            if (checkOffset < 0)
            {
                checkOffset = Mathf.Abs(map.uniqueID * 37) % SeagullMath.CheckInterval;
            }
            if ((Find.TickManager.TicksGame + checkOffset) % SeagullMath.CheckInterval != 0)
            {
                return;
            }
            if (!DepthRimModSettings.Current.seagulls || Find.CurrentMap != map || !Applies)
            {
                return;
            }
            if (!WeatherAllows() || SeagullMath.Room(gulls.Count) == 0)
            {
                return;
            }
            if (Rand.Chance(SeagullMath.ChancePerCheck()))
            {
                TrySpawnFlock();
            }
        }

        public bool WeatherAllows()
        {
            Map skyMap = SeaLevels.SeaLevelOf(map) ?? map;
            float sway = map.GetComponent<MapComponent_SeaSurface>()?.TargetSwayFactor() ?? 1f;
            return SeagullMath.WeatherAllows(skyMap.skyManager.CurSkyGlow, sway);
        }

        // ── 落腳格 ───────────────────────────────────────────────────────────

        /// <summary>這一格能不能落、怎麼落（地形、建築、屋頂、人、已經有別隻要落的格子）。</summary>
        public SeagullPerch PerchAt(IntVec3 c)
        {
            if (!c.InBounds(map) || c.Fogged(map))
            {
                return SeagullPerch.None;
            }
            bool platformLevel = SeaLevels.IsPlatformLevel(map);
            TerrainDef t = map.terrainGrid.TerrainAt(c);
            bool deck = t == DepthRimDefOf.DepthRim_OceanPlatform;
            bool openSea = t == DepthRimDefOf.DepthRim_OpenSea;
            bool hull = !openSea && t?.GetModExtension<HullPlatformPropsExtension>()?.FloatsOnSea == true;
            bool above = false;
            if (!platformLevel)
            {
                Map platform = SeaLevels.PlatformLevelOf(map);
                above = platform != null && c.InBounds(platform) && platform.terrainGrid.TerrainAt(c) == DepthRimDefOf.DepthRim_OceanPlatform;
            }
            bool occupied = TakenByGull(c) || c.GetFirstPawn(map) != null;
            return SeagullMath.PerchFor(platformLevel, deck, openSea, hull, above, c.GetEdifice(map) != null, c.Roofed(map), occupied);
        }

        private bool TakenByGull(IntVec3 c)
        {
            for (int i = 0; i < gulls.Count; i++)
            {
                if (gulls[i].cell == c && gulls[i].state != SeagullState.Departing)
                {
                    return true;
                }
            }
            return false;
        }

        /// <summary>落腳格四周有沒有人（嚇飛）。</summary>
        public bool Disturbed(IntVec3 c)
        {
            int n = GenRadial.NumCellsInRadius(SeagullMath.DisturbRadius);
            for (int i = 0; i < n; i++)
            {
                IntVec3 x = c + GenRadial.RadialPattern[i];
                if (x.InBounds(map) && x.GetFirstPawn(map) != null)
                {
                    return true;
                }
            }
            return false;
        }

        /// <summary>
        /// 挑一個落腳格：優先挑鏡頭畫面裡的（看得到才有意義），試 <see cref="SeagullMath.CellTries"/> 次。
        /// <paramref name="near"/> 有效時只在它附近挑（一小群的同伴）。
        /// </summary>
        public bool TryFindPerch(out IntVec3 cell, IntVec3? near = null)
        {
            CellRect area;
            if (near.HasValue)
            {
                area = CellRect.CenteredOn(near.Value, Mathf.CeilToInt(SeagullMath.FlockRadius));
            }
            else if (Find.CurrentMap == map && Find.CameraDriver != null)
            {
                area = Find.CameraDriver.CurrentViewRect.ExpandedBy(-2);
            }
            else
            {
                area = CellRect.WholeMap(map);
            }
            area = area.ClipInsideMap(map);
            if (area.Area <= 0)
            {
                area = CellRect.WholeMap(map);
            }
            for (int i = 0; i < SeagullMath.CellTries; i++)
            {
                IntVec3 c = area.RandomCell;
                if (PerchAt(c) != SeagullPerch.None && !Disturbed(c))
                {
                    cell = c;
                    return true;
                }
            }
            cell = IntVec3.Invalid;
            return false;
        }

        // ── 生成 ─────────────────────────────────────────────────────────────

        /// <summary>來一隻，必要時再來一兩隻同伴（數量上限內）。回傳來了幾隻。</summary>
        public int TrySpawnFlock()
        {
            if (SeagullMath.Room(gulls.Count) == 0 || !TryFindPerch(out IntVec3 first))
            {
                return 0;
            }
            int wanted = Rand.Chance(SeagullMath.FlockChance) ? Rand.RangeInclusive(2, 3) : 1;
            wanted = Mathf.Min(wanted, SeagullMath.Room(gulls.Count));
            float angle = Rand.Range(0f, Mathf.PI * 2f);
            int spawned = 0;
            if (SpawnAt(first, angle, 0) != null)
            {
                spawned++;
            }
            for (int i = 1; i < wanted; i++)
            {
                if (TryFindPerch(out IntVec3 mate, first) && SpawnAt(mate, angle + Rand.Range(-0.3f, 0.3f), i * Rand.RangeInclusive(20, 45)) != null)
                {
                    spawned++;
                }
            }
            return spawned;
        }

        /// <summary>
        /// 叫一隻海鷗飛來 <paramref name="c"/>（驗收測試、除錯也用；不檢查天氣與上限）。
        /// <paramref name="heading"/> 是飛來的方向（弧度），<paramref name="delayTicks"/> 讓同伴晚一點到。
        /// </summary>
        public Seagull SpawnAt(IntVec3 c, float? heading = null, int delayTicks = 0)
        {
            SeagullPerch perch = PerchAt(c);
            if (perch == SeagullPerch.None)
            {
                return null;
            }
            if (!DefDatabase<SeagullDef>.AllDefsListForReading.TryRandomElementByWeight(d => d.Usable ? d.weight : 0f, out SeagullDef def))
            {
                return null;
            }
            float a = heading ?? Rand.Range(0f, Mathf.PI * 2f);
            Vector2 dir = new Vector2(Mathf.Cos(a), Mathf.Sin(a));
            Vector2 perchPos = new Vector2(c.x + 0.5f + Rand.Range(-0.2f, 0.2f), c.z + 0.5f + Rand.Range(-0.15f, 0.15f));
            float dist = SeagullMath.DistanceToLeaveView(perchPos, -dir, ViewMin(), ViewMax());
            Seagull g = new Seagull
            {
                def = def,
                state = SeagullState.Arriving,
                cell = c,
                perch = perch,
                perchPos = perchPos,
                from = perchPos - dir * dist,
                to = perchPos,
                legTicks = SeagullMath.TravelTicks(dist),
                stateAge = -Mathf.Max(0, delayTicks),
                restTicks = Rand.RangeInclusive(SeagullMath.MinRestTicks, SeagullMath.MaxRestTicks),
                facing = dir.x >= 0f ? Rot4.East : Rot4.West,
                phase = Rand.Range(0, 1000),
            };
            gulls.Add(g);
            totalArrivals++;
            return g;
        }

        private Vector2 ViewMin()
        {
            if (Find.CurrentMap == map && Find.CameraDriver != null)
            {
                CellRect r = Find.CameraDriver.CurrentViewRect;
                return new Vector2(r.minX, r.minZ);
            }
            return Vector2.zero;
        }

        private Vector2 ViewMax()
        {
            if (Find.CurrentMap == map && Find.CameraDriver != null)
            {
                CellRect r = Find.CameraDriver.CurrentViewRect;
                return new Vector2(r.maxX + 1, r.maxZ + 1);
            }
            return new Vector2(map.Size.x, map.Size.z);
        }

        public void ClearAll()
        {
            gulls.Clear();
        }

        /// <summary>叫這隻飛走（停留時間到、被嚇到、落腳處沒了）。</summary>
        public void TakeOff(Seagull g, bool startled = false)
        {
            if (g.state == SeagullState.Departing || g.state == SeagullState.Gone)
            {
                return;
            }
            Vector2 start = g.GroundPos;
            float height = g.Height;
            float a = Rand.Range(0f, Mathf.PI * 2f);
            Vector2 dir = new Vector2(Mathf.Cos(a), Mathf.Sin(a));
            float dist = SeagullMath.DistanceToLeaveView(start, dir, ViewMin(), ViewMax());
            g.state = SeagullState.Departing;
            g.from = start;
            g.to = start + dir * dist;
            g.legTicks = SeagullMath.TravelTicks(dist);
            // 還在空中（飛來途中就被嚇到）：從目前的高度接著爬升，不會先掉回地面
            g.stateAge = height > 0f ? Mathf.RoundToInt(SeagullMath.ClimbShare * g.legTicks * Mathf.Pow(height / SeagullMath.CruiseHeight, 2f)) : 0;
            g.startled = startled;
            if (startled)
            {
                totalStartled++;
            }
        }

        // ── 更新 ─────────────────────────────────────────────────────────────

        private void Advance()
        {
            for (int i = gulls.Count - 1; i >= 0; i--)
            {
                Seagull g = gulls[i];
                g.stateAge++;
                switch (g.state)
                {
                    case SeagullState.Arriving:
                        if (g.stateAge >= g.legTicks)
                        {
                            if (PerchStillFree(g) && !Disturbed(g.cell))
                            {
                                g.state = SeagullState.Resting;
                                g.stateAge = 0;
                            }
                            else
                            {
                                TakeOff(g, startled: true);
                            }
                        }
                        break;
                    case SeagullState.Resting:
                        if (g.stateAge >= g.restTicks)
                        {
                            TakeOff(g);
                        }
                        else if (g.stateAge % SeagullMath.RestCheckInterval == 0)
                        {
                            if (!PerchStillFree(g) || Disturbed(g.cell))
                            {
                                TakeOff(g, startled: true);
                            }
                            else if (Rand.Chance(0.04f))
                            {
                                // 偶爾轉個身
                                g.facing = Rand.Element(Rot4.East, Rot4.West, Rot4.East, Rot4.West, Rot4.South, Rot4.North);
                            }
                        }
                        break;
                    case SeagullState.Departing:
                        if (g.stateAge >= g.legTicks)
                        {
                            g.state = SeagullState.Gone;
                        }
                        break;
                }
                if (g.state == SeagullState.Gone || g.def == null)
                {
                    gulls.RemoveAt(i);
                }
            }
        }

        /// <summary>落腳格的地形還能落（建築、屋頂、地形改變後不能落）；自己不算佔用。</summary>
        private bool PerchStillFree(Seagull g)
        {
            IntVec3 c = g.cell;
            if (!c.InBounds(map))
            {
                return false;
            }
            bool platformLevel = SeaLevels.IsPlatformLevel(map);
            TerrainDef t = map.terrainGrid.TerrainAt(c);
            bool deck = t == DepthRimDefOf.DepthRim_OceanPlatform;
            bool openSea = t == DepthRimDefOf.DepthRim_OpenSea;
            bool hull = !openSea && t?.GetModExtension<HullPlatformPropsExtension>()?.FloatsOnSea == true;
            return SeagullMath.PerchFor(platformLevel, deck, openSea, hull, false, c.GetEdifice(map) != null, c.Roofed(map), false) == g.perch;
        }
    }

    /// <summary>
    /// 海鷗的畫面（每張地圖一份，原版用反射建立；不是海面地圖時 <see cref="Visible"/> 是 false）。每幀直接畫：
    /// 停著的海鷗畫在 Pawn 高度、飛著的畫在天空高度（往北偏離地高度），地上留一個隨高度縮小的影子；浮在海面的會隨海水擺動（<see cref="SeaSway"/>）。
    /// </summary>
    public class MapDrawLayer_Seagulls : MapDrawLayer
    {
        private static Material shadowMat;

        private MapComponent_Seagulls comp;

        /// <summary>最近一幀畫出了幾隻（給驗收測試）。</summary>
        public int LastDrawnCount { get; private set; }

        public MapDrawLayer_Seagulls(Map map) : base(map)
        {
        }

        private MapComponent_Seagulls Comp => comp ?? (comp = Map.GetComponent<MapComponent_Seagulls>());

        public override bool Visible
        {
            get
            {
                MapComponent_Seagulls c = Comp;
                return c != null && c.Count > 0 && !GravshipCapturer.IsGravshipRenderInProgress;
            }
        }

        public override void Regenerate()
        {
            Dirty = false;
        }

        private static Material ShadowMat => shadowMat ?? (shadowMat = MaterialPool.MatFrom("Things/Skyfaller/SkyfallerShadowCircle", ShaderDatabase.Transparent, new Color(1f, 1f, 1f, 0.45f)));

        public override void DrawLayer()
        {
            LastDrawnCount = 0;
            MapComponent_Seagulls c = Comp;
            if (c == null || c.Count == 0 || !Visible)
            {
                return;
            }
            float sway = Map.GetComponent<MapComponent_SeaSurface>()?.SwayFactor ?? 1f;
            float seconds = Find.TickManager.TicksGame / 60f;
            IReadOnlyList<Seagull> gulls = c.Gulls;
            for (int i = 0; i < gulls.Count; i++)
            {
                Seagull g = gulls[i];
                if (g.def == null || !g.def.Usable || g.stateAge < 0)
                {
                    continue;
                }
                if (Draw(g, sway, seconds))
                {
                    LastDrawnCount++;
                }
            }
        }

        private bool Draw(Seagull g, float sway, float seconds)
        {
            Vector2 ground = g.GroundPos;
            float height = g.Height;
            Mesh mesh;
            Material mat;
            Vector3 pos;
            if (g.Flying)
            {
                Vector2 dir = g.to - g.from;
                int facing = SeagullMath.FacingIndex(dir);
                int frame = SeagullMath.FlightFrame(g.stateAge + g.phase, g.def.flyingFrameCount, g.def.flyingTicksPerFrame, g.Gliding);
                Graphic graphic = g.def.FlyingFrame(frame, facing <= 1 ? 0 : facing - 1);
                if (graphic == null)
                {
                    return false;
                }
                Vector2 size = graphic.drawSize;
                mesh = facing == 1 ? MeshPool.GridPlaneFlip(size) : MeshPool.GridPlane(size);
                mat = graphic.MatSingle;
                // 快落地時降到 Pawn 高度，蓋在甲板與建築上；高空時在天空層
                float alt = height > 0.6f ? AltitudeLayer.Skyfaller.AltitudeFor() : AltitudeLayer.Pawn.AltitudeFor();
                pos = new Vector3(ground.x, alt, ground.y + height);
                float shadow = SeagullMath.ShadowScale(height) * g.def.standDrawSize * 0.8f;
                if (g.perch == SeagullPerch.Stand || height > 0.3f)
                {
                    DrawShadow(ground, shadow);
                }
            }
            else
            {
                Graphic graphic = g.perch == SeagullPerch.Float ? g.def.FloatGraphic : g.def.StandGraphic;
                if (graphic == null)
                {
                    return false;
                }
                Vector2 at = ground;
                if (g.perch == SeagullPerch.Float)
                {
                    // 跟著海水擺：海面花紋的位移（SeaSway.Offset）加一點自己的上下浮動
                    at += SeaSway.Offset(seconds, SeaSway.BaseAmplitude * sway) * 0.5f;
                    at.y += 0.04f * Mathf.Sin(seconds * 2.1f + g.phase);
                }
                else
                {
                    DrawShadow(at + new Vector2(0f, -0.12f), g.def.standDrawSize * 0.55f);
                }
                mesh = graphic.MeshAt(g.facing);
                mat = graphic.MatAt(g.facing);
                pos = new Vector3(at.x, AltitudeLayer.Pawn.AltitudeFor(), at.y);
            }
            if (mat == null)
            {
                return false;
            }
            Graphics.DrawMesh(mesh, pos, Quaternion.identity, mat, 0);
            return true;
        }

        private static void DrawShadow(Vector2 at, float size)
        {
            Material m = ShadowMat;
            if (m == null || size <= 0.01f)
            {
                return;
            }
            Vector3 pos = new Vector3(at.x, AltitudeLayer.Shadows.AltitudeFor(), at.y);
            Matrix4x4 matrix = Matrix4x4.TRS(pos, Quaternion.identity, new Vector3(size, 1f, size * 0.6f));
            Graphics.DrawMesh(MeshPool.plane10, matrix, m, 0);
        }
    }

    public static class DebugActions_Seagulls
    {
        [DebugAction("DepthRim", "Seagull lands here", actionType = DebugActionType.ToolMap, allowedGameStates = AllowedGameStates.PlayingOnMap)]
        public static void SeagullHere()
        {
            Map map = Find.CurrentMap;
            MapComponent_Seagulls comp = map?.GetComponent<MapComponent_Seagulls>();
            if (comp == null || !comp.Applies || comp.SpawnAt(UI.MouseCell()) == null)
            {
                Messages.Message("DepthRim.Seagull.CannotLand".Translate(), MessageTypeDefOf.RejectInput, historical: false);
            }
        }

        [DebugAction("DepthRim", "Seagull flock arrives", allowedGameStates = AllowedGameStates.PlayingOnMap)]
        public static void SeagullFlock()
        {
            MapComponent_Seagulls comp = Find.CurrentMap?.GetComponent<MapComponent_Seagulls>();
            if (comp == null || !comp.Applies || comp.TrySpawnFlock() == 0)
            {
                Messages.Message("DepthRim.Seagull.CannotLand".Translate(), MessageTypeDefOf.RejectInput, historical: false);
            }
        }
    }
}
````

````csharp
using System.IO;

// 海鷗（海面兩層的裝飾性生態演出）：生成頻率與上限、能落的格子、飛行高度與路線、影格，以及內容。
static partial class Test
{
    static void SeagullMathTests()
    {
        // 頻率：平均 2 遊戲小時一次，每 250 tick 判斷約 4.9%
        float p = SeagullMath.ChancePerCheck();
        Check(p > 0.045f && p < 0.052f, $"about 4.9% per 250-tick check ({p:P2})");
        float expectedTicks = SeagullMath.CheckInterval / p;
        Check(expectedTicks > 4500f && expectedTicks < 5500f, $"on average one arrival every ~5000 ticks ({expectedTicks:F0})");
        Check(SeagullMath.ChancePerCheck(0f) == 1f, "a zero MTB always fires");
        Check(SeagullMath.ChancePerCheck(1f) > SeagullMath.ChancePerCheck(4f), "a shorter MTB fires more often");

        // 上限
        Check(SeagullMath.MaxPerMap == 5, "at most five seagulls per level");
        Check(SeagullMath.Room(0) == 5 && SeagullMath.Room(4) == 1 && SeagullMath.Room(5) == 0 && SeagullMath.Room(9) == 0, "room left under the cap");

        // 天氣：夜裡、暴風雨不來
        Check(SeagullMath.WeatherAllows(1f, 1f), "a calm day is fine");
        Check(!SeagullMath.WeatherAllows(0.1f, 1f), "not at night");
        Check(!SeagullMath.WeatherAllows(1f, SeaSway.WeatherFactor(2f)), "not in a storm");
        Check(SeagullMath.WeatherAllows(1f, SeaSway.WeatherFactor(0.8f)), "a breeze is fine");

        // 能落的格子
        // 參數：platformLevel, deck, openSea, floatingHull, platformAbove, edifice, roofed, occupied
        Check(SeagullMath.PerchFor(true, true, false, false, false, false, false, false) == SeagullPerch.Stand, "platform top: stands on the deck");
        Check(SeagullMath.PerchFor(true, false, false, false, false, false, false, false) == SeagullPerch.None, "platform level: not in the open air");
        Check(SeagullMath.PerchFor(true, true, false, false, false, true, false, false) == SeagullPerch.None, "platform top: not on a building");
        Check(SeagullMath.PerchFor(true, true, false, false, false, false, true, false) == SeagullPerch.None, "platform top: not under a roof");
        Check(SeagullMath.PerchFor(true, true, false, false, false, false, false, true) == SeagullPerch.None, "platform top: not on someone");
        Check(SeagullMath.PerchFor(false, false, true, false, false, false, false, false) == SeagullPerch.Float, "water level: floats on open sea");
        Check(SeagullMath.PerchFor(false, false, false, true, false, false, false, false) == SeagullPerch.Stand, "water level: stands on a floating hull");
        Check(SeagullMath.PerchFor(false, false, true, false, true, false, false, false) == SeagullPerch.None, "water level: not in the shadow under a platform");
        Check(SeagullMath.PerchFor(false, false, true, false, false, true, false, false) == SeagullPerch.None, "water level: not on a pillar");
        Check(SeagullMath.PerchFor(false, true, false, false, false, false, false, false) == SeagullPerch.None, "water level: an old raised deck is not a perch");

        // 高度：巡航 → 滑翔落地；起飛爬升 → 巡航
        Near(SeagullMath.ArrivalHeight(0f), SeagullMath.CruiseHeight, "arrives at cruise height");
        Near(SeagullMath.ArrivalHeight(1f - SeagullMath.GlideShare), SeagullMath.CruiseHeight, "glide starts at cruise height");
        Near(SeagullMath.ArrivalHeight(1f), 0f, "lands at height 0");
        bool descends = true;
        for (float t = 1f - SeagullMath.GlideShare; t < 1f; t += 0.02f)
        {
            if (SeagullMath.ArrivalHeight(t + 0.02f) > SeagullMath.ArrivalHeight(t) + 1e-5f) { descends = false; }
        }
        Check(descends, "the glide only goes down");
        Check(SeagullMath.ArrivalHeight(0.98f) < 0.05f * SeagullMath.CruiseHeight, "slows down just before touching down");
        Near(SeagullMath.DepartureHeight(0f), 0f, "takes off from the ground");
        Near(SeagullMath.DepartureHeight(SeagullMath.ClimbShare), SeagullMath.CruiseHeight, "reaches cruise height");
        Near(SeagullMath.DepartureHeight(1f), SeagullMath.CruiseHeight, "leaves at cruise height");
        Check(SeagullMath.DepartureHeight(0.05f) > 0.3f * SeagullMath.CruiseHeight, "climbs quickly at first");

        // 路線：從畫面外飛來、飛到畫面外
        Vector2 vmin = new Vector2(10, 10), vmax = new Vector2(50, 40);
        Vector2 perch = new Vector2(30, 25);
        foreach (Vector2 dir in new[] { Vector2.right, Vector2.left, Vector2.up, Vector2.down, new Vector2(1, 1).normalized, new Vector2(-1, -0.3f).normalized })
        {
            float d = SeagullMath.DistanceToLeaveView(perch, dir, vmin, vmax);
            Vector2 end = perch + dir * d;
            bool outside = end.x < vmin.x || end.x > vmax.x || end.y < vmin.y || end.y > vmax.y;
            Check(outside && d >= SeagullMath.MinFlightDistance && d <= SeagullMath.MaxFlightDistance, $"flies out of view along {dir} ({d:F1} cells, ends at {end})");
        }
        // 往南飛出畫面時，身體畫在地面位置北方巡航高度處，也要出畫面
        float south = SeagullMath.DistanceToLeaveView(new Vector2(30, 30), Vector2.down, vmin, new Vector2(50, 100));
        Check(30f - south + SeagullMath.CruiseHeight < vmin.y, $"heading south, the drawn body (above the shadow) leaves the view too ({south:F1})");
        Check(SeagullMath.DistanceToLeaveView(perch, Vector2.right, new Vector2(0, 0), new Vector2(1000, 1000)) == SeagullMath.MaxFlightDistance, "a huge view is capped");
        Check(SeagullMath.DistanceToLeaveView(perch, Vector2.zero, vmin, vmax) == SeagullMath.MinFlightDistance, "no direction falls back to the minimum");
        Check(SeagullMath.TravelTicks(18f) == 150, "18 cells at 0.12 cells/tick take 150 ticks");
        Check(SeagullMath.TravelTicks(0f) == 1, "never zero ticks");

        // 影格與方向
        Check(SeagullMath.FlightFrame(0, 8, 2, false) == 0 && SeagullMath.FlightFrame(3, 8, 2, false) == 1 && SeagullMath.FlightFrame(16, 8, 2, false) == 0, "flaps through 8 frames, 2 ticks each");
        Check(SeagullMath.FlightFrame(3, 8, 2, true) == 0 && SeagullMath.FlightFrame(4, 8, 2, true) == 1, "flaps half as fast while gliding");
        Check(SeagullMath.FlightFrame(-5, 8, 2, false) >= 0, "never a negative frame");
        Check(SeagullMath.FlightFrame(10, 0, 2, false) == 0, "no frames, frame 0");
        Check(SeagullMath.FacingIndex(new Vector2(1, 0.2f)) == 0 && SeagullMath.FacingIndex(new Vector2(-1, 0.5f)) == 1, "sideways flight shows the side (west is flipped east)");
        Check(SeagullMath.FacingIndex(new Vector2(0.1f, 1)) == 2 && SeagullMath.FacingIndex(new Vector2(0.1f, -1)) == 3, "straight north / south");
        Check(SeagullMath.ShadowScale(0f) == 1f && SeagullMath.ShadowScale(SeagullMath.CruiseHeight) < 0.6f, "the shadow shrinks with height");

        // 停留時間與嚇飛範圍合理
        Check(SeagullMath.MinRestTicks >= 1200 && SeagullMath.MaxRestTicks > SeagullMath.MinRestTicks, "rests for a while");
        Check(SeagullMath.DisturbRadius >= 2f && SeagullMath.DisturbRadius <= 5f, "people a few cells away scare it off");
    }
````

````csharp
using System.Collections.Generic;
using System.Linq;
using RimWorld;
using Verse;

namespace DepthRim.Acceptance
{
    /// <summary>海鷗：海面兩層的裝飾性生態演出（純畫面，不是 Pawn）。</summary>
    public static class Tests_Seagulls
    {
        /// <summary>
        /// K2：水面層開一塊浮船、蓋支柱建立平台層。海鷗能浮在開放海面、站在浮船與平台頂甲板上，不能落在高空、支柱、平台陰影下；
        /// 飛來後停下、被靠近的人嚇飛、飛走後消失；一次一小群不超過上限；兩層都畫得出來並各截一張圖。
        /// 測試期間關掉自然生成（模組設定），只用明確叫來的海鷗。
        /// </summary>
        public static IEnumerable<Wait> K2_Seagulls(TestContext ctx)
        {
            bool settingBefore = DepthRimModSettings.Current.seagulls;
            DepthRimModSettings.Current.seagulls = false;
            Map sea = Tests_SeaSurface.CreateSeaMap(ctx);
            IntVec3 center = sea.Center;
            // 長條浮船當碼頭：殖民者站在西端，東端離他夠遠
            CellRect dock = new CellRect(center.x - 8, center.z, 17, 3);
            Tests_SeaSurface.LayHull(sea, dock);
            Pawn colonist = DepthRimFixture.SpawnColonist(sea, new IntVec3(dock.minX, 0, dock.minZ + 1));
            yield return new WaitFrames(5);
            ctx.Require(sea.Parent.HasMap, "the sea surface map still exists");
            IntVec3 p1 = new IntVec3(center.x, 0, dock.minZ - 5);
            Building_PillarFoot foot = Tests_SeaSurface.BuildPillar(sea, p1);
            yield return new WaitFrames(3);
            Map platform = SeaLevels.PlatformLevelOf(sea);
            ctx.Require(platform != null && foot.Top != null, "the pillar created the platform level");

            MapComponent_Seagulls seaGulls = sea.GetComponent<MapComponent_Seagulls>();
            MapComponent_Seagulls topGulls = platform.GetComponent<MapComponent_Seagulls>();
            ctx.Require(seaGulls != null && topGulls != null, "both levels have a seagull component");
            ctx.Check(seaGulls.Applies && topGulls.Applies, "seagulls apply to both levels");
            MapComponent_Seagulls homeGulls = DepthRimFixture.Home.GetComponent<MapComponent_Seagulls>();
            ctx.Check(homeGulls != null && !homeGulls.Applies, "but not to the home map");
            ctx.Check(DefDatabase<SeagullDef>.AllDefsListForReading.Any(d => d.Usable), "a seagull look with all textures loaded (Odyssey swan stand-in)");

            // 能落的格子
            IntVec3 farSea = new IntVec3(center.x + 20, 0, center.z + 20);
            IntVec3 hullFar = new IntVec3(dock.maxX, 0, dock.minZ + 1);
            IntVec3 underDeck = p1 + new IntVec3(2, 0, 2);
            ctx.Check(seaGulls.PerchAt(farSea) == SeagullPerch.Float, "water level: floats on open sea");
            ctx.Check(seaGulls.PerchAt(hullFar) == SeagullPerch.Stand, "water level: stands on a floating hull");
            ctx.Check(seaGulls.PerchAt(p1) == SeagullPerch.None, "water level: not on the pillar");
            ctx.Check(platform.terrainGrid.TerrainAt(underDeck) == DepthRimDefOf.DepthRim_OceanPlatform && seaGulls.PerchAt(underDeck) == SeagullPerch.None, "water level: not in the shade right under the platform deck");
            ctx.Check(seaGulls.PerchAt(colonist.Position) == SeagullPerch.None, "not on a colonist");
            List<IntVec3> deckCells = platform.AllCells.Where(c => platform.terrainGrid.TerrainAt(c) == DepthRimDefOf.DepthRim_OceanPlatform).ToList();
            IntVec3 deckFree = deckCells.FirstOrDefault(c => topGulls.PerchAt(c) == SeagullPerch.Stand);
            ctx.Require(deckFree.IsValid && deckCells.Contains(deckFree), "platform top: a free deck cell to land on");
            ctx.Check(topGulls.PerchAt(p1) == SeagullPerch.None, "platform top: not on the pillar top");
            ctx.Check(topGulls.PerchAt(farSea) == SeagullPerch.None, "platform level: not in the open air");

            // 平台頂：飛來 → 停下 → 畫得出來
            Current.Game.CurrentMap = platform;
            Seagull top = topGulls.SpawnAt(deckFree, 0.4f);
            ctx.Require(top != null && top.state == SeagullState.Arriving && top.perch == SeagullPerch.Stand, "a seagull flies towards the platform top");
            ctx.Check(top.Height > 3f, $"it arrives high up ({top.Height:F1} cells)");
            ctx.Check(top.legTicks >= SeagullMath.TravelTicks(SeagullMath.MinFlightDistance), $"from outside the view ({top.legTicks} ticks)");
            ctx.Check(topGulls.SpawnAt(deckFree) == null, "a second one cannot take the same cell");
            WaitUntil landed = new WaitUntil(() => top.state == SeagullState.Resting, top.legTicks + 120);
            yield return landed;
            ctx.Check(!landed.TimedOut && top.Height == 0f, $"it lands on the deck (state {top.state})");
            foreach (Wait w in ctx.ScreenshotOn(platform, deckFree, 8f, "seagull_platform_top", 30)) { yield return w; }
            MapDrawLayer_Seagulls topLayer = platform.mapDrawer.GetGlobalLayer<MapDrawLayer_Seagulls>();
            ctx.Check(topLayer != null && topLayer.LastDrawnCount == 1, $"the platform level draws it ({topLayer?.LastDrawnCount})");
            ctx.Check(top.state == SeagullState.Resting, "still resting a moment later");

            // 有人走近就嚇飛，飛走後消失
            Pawn walker = DepthRimFixture.SpawnColonist(platform, deckCells.First(c => c != deckFree && c != p1 && c.Standable(platform) && c.InHorDistOf(deckFree, SeagullMath.DisturbRadius)));
            WaitUntil scared = new WaitUntil(() => top.state != SeagullState.Resting, SeagullMath.RestCheckInterval + 10);
            yield return scared;
            ctx.Check(!scared.TimedOut && top.state == SeagullState.Departing && top.startled && topGulls.totalStartled == 1, $"someone walking close scares it off (state {top.state})");
            WaitUntil gone = new WaitUntil(() => topGulls.Count == 0, top.legTicks + 60);
            yield return gone;
            ctx.Check(!gone.TimedOut, "it flies away and disappears");
            walker.DeSpawn();

            // 水面層：浮在海上的與站在浮船上的
            Current.Game.CurrentMap = sea;
            Seagull floater = seaGulls.SpawnAt(farSea);
            Seagull stander = seaGulls.SpawnAt(hullFar);
            ctx.Require(floater != null && stander != null, "seagulls come to the water level");
            ctx.Check(floater.perch == SeagullPerch.Float && stander.perch == SeagullPerch.Stand, "one floats, one stands on the hull");
            yield return new WaitUntil(() => floater.state == SeagullState.Resting && stander.state == SeagullState.Resting, System.Math.Max(floater.legTicks, stander.legTicks) + 120);
            ctx.Check(floater.state == SeagullState.Resting && stander.state == SeagullState.Resting, "both landed");
            foreach (Wait w in ctx.ScreenshotOn(sea, hullFar + new IntVec3(6, 0, 4), 14f, "seagull_water_level", 30)) { yield return w; }
            MapDrawLayer_Seagulls seaLayer = sea.mapDrawer.GetGlobalLayer<MapDrawLayer_Seagulls>();
            ctx.Check(seaLayer != null && seaLayer.LastDrawnCount >= 1, $"the water level draws them ({seaLayer?.LastDrawnCount})");

            // 浮船被拆掉（地形變成海）：站著的那隻落腳處沒了，飛走
            sea.terrainGrid.SetTerrain(hullFar, DepthRimDefOf.DepthRim_OpenSea);
            yield return new WaitUntil(() => stander.state != SeagullState.Resting, SeagullMath.RestCheckInterval + 10);
            ctx.Check(stander.state == SeagullState.Departing, "a seagull whose perch disappears takes off");

            // 一小群：上限內
            seaGulls.ClearAll();
            int arrived = 0;
            for (int i = 0; i < 12; i++)
            {
                arrived += seaGulls.TrySpawnFlock();
            }
            ctx.Check(arrived > 0 && seaGulls.Count <= SeagullMath.MaxPerMap, $"flocks never exceed the cap ({seaGulls.Count} of {SeagullMath.MaxPerMap}, {arrived} arrived)");
            ctx.Check(seaGulls.Count == SeagullMath.MaxPerMap, "and fill it when called often");
            ctx.Note($"sky glow {sea.skyManager.CurSkyGlow:F2}, weather allows seagulls: {seaGulls.WeatherAllows()}; chance per check {SeagullMath.ChancePerCheck():P1}");
            seaGulls.ClearAll();
            topGulls.ClearAll();

            DepthRimModSettings.Current.seagulls = settingBefore;
            Tests_SeaSurface.RemoveSeaMaps(sea);
            yield return new WaitFrames(30);
        }
    }
````
